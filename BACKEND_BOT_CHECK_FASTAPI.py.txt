RPC_PUBLIC_ORDER_RATE = {}


def rpc_get_client_ip(request: Request) -> str:
    forwarded = request.headers.get("x-forwarded-for", "")
    if forwarded:
        return forwarded.split(",")[0].strip()
    if request.client:
        return request.client.host
    return "unknown"


def rpc_rate_limit_public_order(ip: str, limit: int = 6, window_seconds: int = 60):
    now = time.time()
    bucket = RPC_PUBLIC_ORDER_RATE.get(ip, [])
    bucket = [t for t in bucket if now - t < window_seconds]

    if len(bucket) >= limit:
        RPC_PUBLIC_ORDER_RATE[ip] = bucket
        raise HTTPException(
            status_code=429,
            detail="Слишком много заявок. Попробуйте позже."
        )

    bucket.append(now)
    RPC_PUBLIC_ORDER_RATE[ip] = bucket


def rpc_verify_turnstile(token: str, request: Request):
    secret = os.getenv("TURNSTILE_SECRET_KEY", "").strip()
    if not secret:
        print("[RPC BOT CHECK] TURNSTILE_SECRET_KEY missing")
        raise HTTPException(status_code=500, detail="Антибот не настроен на сервере")

    if not token:
        raise HTTPException(status_code=403, detail="Антибот-проверка не пройдена")

    try:
        response = requests.post(
            "https://challenges.cloudflare.com/turnstile/v0/siteverify",
            data={
                "secret": secret,
                "response": token,
                "remoteip": rpc_get_client_ip(request),
            },
            timeout=10,
        )
        result = response.json()
    except Exception as e:
        print("[RPC BOT CHECK] Turnstile request error:", e)
        raise HTTPException(status_code=502, detail="Ошибка проверки антибота")

    if not result.get("success"):
        print("[RPC BOT CHECK] Turnstile failed:", result)
        raise HTTPException(status_code=403, detail="Антибот-проверка провалена")

    return True


def rpc_clean_public_order(order: dict) -> dict:
    # Убираем технические антибот-поля, чтобы они не попадали в CRM/Telegram.
    cleaned = dict(order)
    for key in (
        "website",
        "started_at",
        "cf-turnstile-response",
        "turnstile_token",
    ):
        cleaned.pop(key, None)
    return cleaned
